+// test/net-compression.test.js — permessage-deflate is actually offered, and the negotiation stays unrefusable.
+//
+// Why: test/net-socket.test.js pins the TCP tuning but nothing covers `perMessageDeflate`. A silent regression
+// (dropping the option, or writing it back as `false`) would leave no trace anywhere in the suite — yet it is the
+// bandwidth switch for the m.public/m.private broadcasts this server sends ten times a second. This boots a real
+// server with a real client and reads the handshake, which is exactly what a player checks in DevTools.
+
+import { test, describe } from 'node:test';
+import assert from 'node:assert/strict';
+import http from 'node:http';
+import { WebSocketServer, WebSocket } from 'ws';
+
+/** Boot a real ws server with `perMessageDeflate`, connect one real client, return the handshake header. */
+async function extensionsOf(perMessageDeflate) {
+  const server = http.createServer();
+  const wss = new WebSocketServer({ noServer: true, perMessageDeflate });
+  server.on('upgrade', (req, socket, head) =>
+    wss.handleUpgrade(req, socket, head, (ws) => wss.emit('connection', ws, req)));
+  await new Promise((r) => server.listen(0, '127.0.0.1', r));
+  const client = new WebSocket(`ws://127.0.0.1:${server.address().port}/ws`);
+  let got = null;
+  client.on('upgrade', (res) => { got = res.headers['sec-websocket-extensions'] || null; });
+  await new Promise((res, rej) => { client.on('open', res); client.on('error', rej); });
+  client.terminate();
+  wss.close();
+  await new Promise((r) => server.close(r));
+  return got;
+}
+
+describe('WS permessage-deflate', () => {
+  test('the option the server uses negotiates compression', async () => {
+    assert.match(await extensionsOf({ threshold: 64 }), /permessage-deflate/);
+  });
+
+  test('server_no_context_takeover is never declared (declaring it can refuse a valid client)', async () => {
+    assert.doesNotMatch(await extensionsOf({ threshold: 64 }), /server_no_context_takeover/);
+  });
+
+  test('the baseline this replaces (perMessageDeflate: false) negotiates nothing', async () => {
+    assert.equal(await extensionsOf(false), null);
+  });
+});
